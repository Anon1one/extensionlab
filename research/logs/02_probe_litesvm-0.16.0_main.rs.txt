struct Ctx {

fn describe(e: &FailedTransactionMetadata) -> String {
    let raw = format!("{:?}", e.err);
    let mut name = String::new();
    if let Some(pos) = raw.find("Custom(") {
        let rest = &raw[pos + 7..];
        if let Some(end) = rest.find(')') {
            if let Ok(n) = rest[..end].parse::<usize>() {
                let t22_failed = e
                    .meta
                    .logs
                    .iter()
                    .any(|l| l.contains("TokenzQdBNbLqP5VEhdkAS6EPFLC1PHnBqCXEpPxuEb failed"));
                if t22_failed && n < TOKEN_ERRORS.len() {
                    name = format!(" => TokenError::{} (0x{:x})", TOKEN_ERRORS[n], n);
                }
            }
        }
    }
    let last: Vec<&String> = e
        .meta
        .logs
        .iter()
        .filter(|l| l.contains("Error") || l.contains("failed") || l.contains("Warning"))
        .collect();
    format!("{raw}{name} | logs: {last:?}")
}

impl Ctx {
    fn new() -> Self {
        let mut svm = LiteSVM::new();
        let payer = Keypair::new();
        svm.airdrop(&payer.pubkey(), 1_000_000_000_000).unwrap();
        Ctx { svm, payer }
    }
    fn p(&self) -> Address {
        self.payer.pubkey()
    }
    fn send(
        &mut self,
        ixs: &[Instruction],
        signers: &[&Keypair],
    ) -> Result<TransactionMetadata, FailedTransactionMetadata> {
        let mut all: Vec<&Keypair> = vec![&self.payer];
        all.extend_from_slice(signers);
        self.svm.expire_blockhash();
        let bh = self.svm.latest_blockhash();
        let tx = Transaction::new(&all, Message::new(ixs, Some(&self.payer.pubkey())), bh);
        self.svm.send_transaction(tx)
    }
    fn ok(&mut self, label: &str, ixs: &[Instruction], signers: &[&Keypair]) -> TransactionMetadata {
        match self.send(ixs, signers) {
            Ok(m) => m,
            Err(e) => panic!("setup step '{label}' failed: {}\n{:#?}", describe(&e), e.meta.logs),
        }
    }
    fn probe(
        &mut self,
        label: &str,
        ixs: &[Instruction],
        signers: &[&Keypair],
    ) -> Result<TransactionMetadata, String> {
        match self.send(ixs, signers) {
            Ok(m) => {
                println!("PROBE {label}: OK");
                Ok(m)
            }
            Err(e) => {
                let d = describe(&e);
                println!("PROBE {label}: FAIL {d}");
                Err(d)
            }
        }
    }
    fn mint_ixs(
        &self,
        mint: &Address,
        exts: &[E],
        pre: Vec<Instruction>,
        decimals: u8,
        freeze: bool,
        extra_rent_bytes: usize,
    ) -> Vec<Instruction> {
        let space = E::try_calculate_account_len::<Mint>(exts).unwrap();
        let lamports = self.svm.minimum_balance_for_rent_exemption(space + extra_rent_bytes);
        let mut v = vec![solana_system_interface::instruction::create_account(
            &self.p(),
            mint,
            lamports,
            space as u64,
            &T22,
        )];
        v.extend(pre);
        let fa = self.p();
        v.push(
            ti::initialize_mint2(&T22, mint, &self.p(), if freeze { Some(&fa) } else { None }, decimals)
                .unwrap(),
        );
        v
    }
    fn mint(&mut self, exts: &[E], pre: Vec<Instruction>, decimals: u8, freeze: bool) -> Keypair {
        let mint = Keypair::new();
        let ixs = self.mint_ixs(&mint.pubkey(), exts, pre, decimals, freeze, 0);
        self.ok("create mint", &ixs, &[&mint]);
        mint
    }
    fn mint_exts(&self, mint: &Address) -> Vec<E> {
        let d = self.svm.get_account(mint).unwrap().data;
        StateWithExtensions::<Mint>::unpack(&d).unwrap().get_extension_types().unwrap()
    }
    fn account_ixs(
        &self,
        acct: &Address,
        mint: &Address,
        owner: &Address,
        extra: &[E],
        pre_init: Vec<Instruction>,
        size_override: Option<usize>,
    ) -> Vec<Instruction> {
        let mut exts = E::get_required_init_account_extensions(&self.mint_exts(mint));
        for e in extra {
            if !exts.contains(e) {
                exts.push(*e);
            }
        }
        let space = size_override
            .unwrap_or_else(|| E::try_calculate_account_len::<Account>(&exts).unwrap());
        let lamports = self.svm.minimum_balance_for_rent_exemption(space);
        let mut v = vec![solana_system_interface::instruction::create_account(
            &self.p(),
            acct,
            lamports,
            space as u64,
            &T22,
        )];
        v.extend(pre_init);
        v.push(ti::initialize_account3(&T22, acct, mint, owner).unwrap());
        v
    }
    fn account(&mut self, mint: &Address, owner: &Address, extra: &[E]) -> Address {
        let k = Keypair::new();
        let pre = if extra.contains(&E::ImmutableOwner) {
            vec![ti::initialize_immutable_owner(&T22, &k.pubkey()).unwrap()]
        } else {
            vec![]
        };
        let ixs = self.account_ixs(&k.pubkey(), mint, owner, extra, pre, None);
        self.ok("create token account", &ixs, &[&k]);
        k.pubkey()
    }
    fn mint_to(&mut self, mint: &Address, to: &Address, amount: u64) {
        let ix = ti::mint_to(&T22, mint, to, &self.p(), &[], amount).unwrap();
        self.ok("mint_to", &[ix], &[]);
    }
    fn bal(&self, a: &Address) -> u64 {
        let d = self.svm.get_account(a).unwrap().data;
        StateWithExtensions::<Account>::unpack(&d).unwrap().base.amount
    }
    fn withheld(&self, a: &Address) -> u64 {
        let d = self.svm.get_account(a).unwrap().data;
        StateWithExtensions::<Account>::unpack(&d)
            .unwrap()
            .get_extension::<TransferFeeAmount>()
            .map(|e| u64::from(e.withheld_amount))
            .unwrap_or(0)
    }
    fn acct_exts(&self, a: &Address) -> Vec<E> {
        let d = self.svm.get_account(a).unwrap().data;
        StateWithExtensions::<Account>::unpack(&d).unwrap().get_extension_types().unwrap()
    }
    fn is_frozen(&self, a: &Address) -> bool {
        let d = self.svm.get_account(a).unwrap().data;
        StateWithExtensions::<Account>::unpack(&d).unwrap().base.is_frozen()
    }
    fn ui(&mut self, mint: &Address, amount: u64) -> String {
        let m = self.ok(
            "amount_to_ui_amount",
            &[ti::amount_to_ui_amount(&T22, mint, amount).unwrap()],
            &[],
        );
        String::from_utf8(m.return_data.data).unwrap()
    }
    fn ui_to_amount(&mut self, mint: &Address, ui: &str) -> u64 {
        let m = self.ok(
            "ui_amount_to_amount",
            &[ti::ui_amount_to_amount(&T22, mint, ui).unwrap()],
            &[],
        );
        u64::from_le_bytes(m.return_data.data[..8].try_into().unwrap())
    }
    fn warp(&mut self, secs: i64, epochs: u64) {
        let mut c: Clock = self.svm.get_sysvar::<Clock>();
        c.unix_timestamp += secs;
        c.epoch += epochs;
        c.slot += 1;
        self.svm.set_sysvar::<Clock>(&c);
    }
    #[allow(clippy::too_many_arguments)]
    fn xfer(&self, src: &Address, mint: &Address, dst: &Address, auth: &Address, amount: u64, decimals: u8) -> Instruction {
        ti::transfer_checked(&T22, src, mint, dst, auth, &[], amount, decimals).unwrap()
    }
}

fn s1_transfer_fee() {
    let mut c = Ctx::new();
    let p = c.p();
    let mint = Keypair::new();
    let m = mint.pubkey();
    let init = transfer_fee::instruction::initialize_transfer_fee_config(&T22, &m, Some(&p), Some(&p), 100, 5_000).unwrap();
    let ixs = c.mint_ixs(&m, &[E::TransferFeeConfig], vec![init], 6, false, 0);
    c.ok("tf mint", &ixs, &[&mint]);
    let alice = Keypair::new();
    let vault_auth = Keypair::new();
    let a = c.account(&m, &alice.pubkey(), &[]);
    let v = c.account(&m, &vault_auth.pubkey(), &[]);
    let b = c.account(&m, &Keypair::new().pubkey(), &[]);
    println!("INFO tf vault account exts {:?}, len {}", c.acct_exts(&v), c.svm.get_account(&v).unwrap().data.len());
    c.mint_to(&m, &a, 10_000);
    let ix = c.xfer(&a, &m, &v, &alice.pubkey(), 10_000, 6);
    c.probe("tf deposit 10000 transfer_checked", &[ix], &[&alice]).ok();
    println!("PROBE tf after deposit: alice={} vault={} vault_withheld={}", c.bal(&a), c.bal(&v), c.withheld(&v));
    let ix = c.xfer(&v, &m, &a, &vault_auth.pubkey(), 10_000, 6);
    c.probe("tf withdraw recorded 10000 from vault", &[ix], &[&vault_auth]).ok();
    let ix = c.xfer(&v, &m, &a, &vault_auth.pubkey(), 9_900, 6);
    c.probe("tf withdraw actual 9900 from vault", &[ix], &[&vault_auth]).ok();
    println!("PROBE tf after withdraw: alice={} vault={} alice_withheld={} vault_withheld={}", c.bal(&a), c.bal(&v), c.withheld(&a), c.withheld(&v));
    let ix = ti::transfer(&T22, &a, &b, &alice.pubkey(), &[], 10).unwrap();
    c.probe("tf legacy Transfer (no mint)", &[ix], &[&alice]).ok();
    let ix = transfer_fee::instruction::transfer_checked_with_fee(&T22, &a, &m, &b, &alice.pubkey(), &[], 1_000, 6, 0).unwrap();
    c.probe("tf transfer_checked_with_fee wrong fee=0", &[ix], &[&alice]).ok();
    let ix = c.xfer(&a, &m, &b, &alice.pubkey(), 1, 6);
    c.probe("tf transfer amount=1", &[ix], &[&alice]).ok();
    println!("PROBE tf after amount=1: bob={} bob_withheld={}", c.bal(&b), c.withheld(&b));
    let ix = ti::close_account(&T22, &v, &p, &vault_auth.pubkey(), &[]).unwrap();
    c.probe("tf close vault with balance 0 and withheld>0", &[ix], &[&vault_auth]).ok();
    let ix = transfer_fee::instruction::harvest_withheld_tokens_to_mint(&T22, &m, &[&v]).unwrap();
    c.probe("tf harvest vault withheld to mint (payer only signs)", &[ix], &[]).ok();
    let ix = ti::close_account(&T22, &v, &p, &vault_auth.pubkey(), &[]).unwrap();
    c.probe("tf close vault after harvest", &[ix], &[&vault_auth]).ok();
    let ix = transfer_fee::instruction::set_transfer_fee(&T22, &m, &p, &[], 500, 5_000).unwrap();
    c.probe("tf set_transfer_fee 500 bps", &[ix], &[]).ok();
    for (label, ep) in [("right after fee change", 0u64), ("at epoch+1", 1), ("at epoch+2", 1)] {
        c.warp(0, ep);
        let before = c.bal(&b);
        let ix = c.xfer(&a, &m, &b, &alice.pubkey(), 1_000, 6);
        c.probe(&format!("tf transfer 1000 {label}"), &[ix], &[&alice]).ok();
        println!("PROBE tf credited for 1000 {label} = {}", c.bal(&b) - before);
    }
    let k = Keypair::new();
    let ixs = c.account_ixs(&k.pubkey(), &m, &p, &[], vec![], Some(165));
    c.probe("tf InitializeAccount3 on 165-byte account", &ixs, &[&k]).ok();
}

    let ixs = c.mint_ixs(&m, &[E::TransferFeeConfig], vec![], 6, false, 0);
    c.probe("order: space for TransferFeeConfig reserved but extension never initialized", &ixs, &[&mint]).ok();
}

/// Close a mint that has one zero-balance holder account, then re-create a
/// mint at the same address with `exts`. Returns (ctx, mint, old_account, old_owner).
fn resurrect(exts: &[E], pre: impl Fn(&Address, &Address) -> Vec<Instruction>, freeze: bool) -> (Ctx, Keypair, Address, Keypair) {
    let mut c = Ctx::new();
    let p = c.p();
    let mint = Keypair::new();
    let m = mint.pubkey();
    let init = ti::initialize_mint_close_authority(&T22, &m, Some(&p)).unwrap();
    let ixs = c.mint_ixs(&m, &[E::MintCloseAuthority], vec![init], 0, false, 0);
    c.ok("mint v1", &ixs, &[&mint]);
    let old_owner = Keypair::new();
    let old = c.account(&m, &old_owner.pubkey(), &[]);
    c.ok("close mint v1", &[ti::close_account(&T22, &m, &p, &p, &[]).unwrap()], &[]);
    let mut all = vec![E::MintCloseAuthority];
    all.extend_from_slice(exts);
    let mut pre_ixs = vec![ti::initialize_mint_close_authority(&T22, &m, Some(&p)).unwrap()];
    pre_ixs.extend(pre(&m, &p));
    let ixs = c.mint_ixs(&m, &all, pre_ixs, 0, freeze, 0);
    c.ok("mint v2 at same address", &ixs, &[&mint]);
    (c, mint, old, old_owner)
}

fn s17_resurrection() {
    // DefaultAccountState(Frozen) added on re-creation: is the old account frozen?
    let (mut c, mint, old, old_owner) = resurrect(
        &[E::DefaultAccountState],
        |m, _| vec![default_account_state::instruction::initialize_default_account_state(&T22, m, &AccountState::Frozen).unwrap()],
        true,
    );
    let m = mint.pubkey();
    let fresh = c.account(&m, &Keypair::new().pubkey(), &[]);
    println!("PROBE res(DAS frozen) old account frozen={} fresh account frozen={}", c.is_frozen(&old), c.is_frozen(&fresh));
    let ix = ti::mint_to(&T22, &m, &old, &c.p(), &[], 100).unwrap();
    c.probe("res(DAS frozen) mint_to old unfrozen account", &[ix], &[]).ok();
    let sink_owner = Keypair::new();
    let sink = c.account(&m, &sink_owner.pubkey(), &[]);
    c.ok("thaw sink", &[ti::thaw_account(&T22, &sink, &m, &c.p(), &[]).unwrap()], &[]);
    let ix = c.xfer(&old, &m, &sink, &old_owner.pubkey(), 100, 0);
    c.probe("res(DAS frozen) old account transfers without ever being thawed", &[ix], &[&old_owner]).ok();

    // NonTransferable added on re-creation
    let (mut c, mint, old, _o) = resurrect(&[E::NonTransferable], |m, _| vec![ti::initialize_non_transferable_mint(&T22, m).unwrap()], false);
    let ix = ti::mint_to(&T22, &mint.pubkey(), &old, &c.p(), &[], 100).unwrap();
    c.probe("res(NonTransferable) mint_to old account", &[ix], &[]).ok();

    // TransferFee added on re-creation: outbound from old account
    let (mut c, mint, old, old_owner) = resurrect(
        &[E::TransferFeeConfig],
        |m, p| vec![transfer_fee::instruction::initialize_transfer_fee_config(&T22, m, Some(p), Some(p), 1000, u64::MAX).unwrap()],
        false,
    );
    let m = mint.pubkey();
    c.mint_to(&m, &old, 1000);
    let dst = c.account(&m, &Keypair::new().pubkey(), &[]);
    let ix = c.xfer(&old, &m, &dst, &old_owner.pubkey(), 1000, 0);
    c.probe("res(TransferFee 10%) transfer 1000 OUT of old account", &[ix], &[&old_owner]).ok();
    println!("PROBE res(TransferFee) dst credited={} dst withheld={}", c.bal(&dst), c.withheld(&dst));

    // TransferHook (non-existent program) added on re-creation: outbound from old account
    let fake = Keypair::new().pubkey();
    let (mut c, mint, old, old_owner) = resurrect(
        &[E::TransferHook],
        move |m, p| vec![transfer_hook::instruction::initialize(&T22, m, Some(*p), Some(fake)).unwrap()],
        false,
    );
    let m = mint.pubkey();
    c.mint_to(&m, &old, 1000);
    let dst = c.account(&m, &Keypair::new().pubkey(), &[]);
    let ix = c.xfer(&old, &m, &dst, &old_owner.pubkey(), 10, 0);
    c.probe("res(TransferHook fake program) transfer OUT of old account, no extra accounts", &[ix], &[&old_owner]).ok();
}

fn s18_permissioned_burn() {
    let mut c = Ctx::new();
    let p = c.p();
    let burn_auth = Keypair::new();
    let mint = Keypair::new();
    let m = mint.pubkey();
    let init = spl_token_2022_interface::extension::permissioned_burn::instruction::initialize(&T22, &m, &burn_auth.pubkey()).unwrap();
    let ixs = c.mint_ixs(&m, &[E::PermissionedBurn], vec![init], 6, false, 0);
    c.ok("pb mint", &ixs, &[&mint]);
    let o = Keypair::new();
    let a = c.account(&m, &o.pubkey(), &[]);
    c.mint_to(&m, &a, 100);
    let ix = ti::burn_checked(&T22, &a, &m, &o.pubkey(), &[], 10, 6).unwrap();
    c.probe("pb owner standard BurnChecked", &[ix], &[&o]).ok();
    let ix = spl_token_2022_interface::extension::permissioned_burn::instruction::burn_checked(&T22, &a, &m, &burn_auth.pubkey(), &o.pubkey(), &[], 10, 6).unwrap();
    c.probe("pb permissioned BurnChecked with burn authority co-sign", &[ix], &[&o, &burn_auth]).ok();
    let _ = p;
}

fn run(name: &str, f: fn()) {
    println!("===== {name}");
    if catch_unwind(AssertUnwindSafe(f)).is_err() {
        println!("SCENARIO {name} PANICKED (see stderr)");
    }
}
